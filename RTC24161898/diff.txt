+
+		WorkItem retorno = null;
+		
+		String query = getRtcQuerySolicitacaoCapex(wi);

+
+            Workitem workitem = RtcService.getWorkitem(query, QTD_BUSCA_RTC, 0, service);

+            if (!workitem.getWorkItem().isEmpty()) {
+                retorno = workitem.getWorkItem().get(0);

+            if (retorno != null) {
+
